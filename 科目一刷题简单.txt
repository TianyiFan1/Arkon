class Solution {
    int calculate(String message) {
        String[] parts = message.split("\\$");  
        long result = 0;
        boolean first = true;

        for (String part : parts) {
            long seg = evalAt(part);
            if (first) {
                result = seg;
                first = false;
            } else {
                result = 3 * result + 2 * seg + 1;  
            }
        }
        return (int) result;
    }

    private long evalAt(String part) {
        String[] nums = part.split("@");
        long acc = Long.parseLong(nums[0]);
        for (int i = 1; i < nums.length; i++) {
            long y = Long.parseLong(nums[i]);
            acc = 2 * acc + y + 3;          
        }
        return acc;
    }
}

import java.util.*;

class Solution {
    int[] getTop3Num(String inputStr) {
        List<Long> nums = new ArrayList<>();
        int n = inputStr.length();
        int i = 0;

        while (i < n) {
            char c = inputStr.charAt(i);
            if (c == '-' && i + 1 < n && Character.isDigit(inputStr.charAt(i + 1))) {
                int j = i + 1;
                while (j < n && Character.isDigit(inputStr.charAt(j))) {
                    j++;
                }
                nums.add(-parseNum(inputStr.substring(i + 1, j)));
                i = j;
            } else if (Character.isDigit(c)) {
                int j = i;
                while (j < n && Character.isDigit(inputStr.charAt(j))) {
                    j++;
                }
                nums.add(parseNum(inputStr.substring(i, j)));
                i = j;
            } else {
                i++;
            }
        }

        nums.sort(Collections.reverseOrder());
        int k = Math.min(3, nums.size());
        int[] res = new int[k];
        for (int t = 0; t < k; t++) {
            res[t] = nums.get(t).intValue();
        }
        return res;
    }

    private long parseNum(String digits) {
        int p = 0;
        while (p < digits.length() - 1 && digits.charAt(p) == '0') {
            p++;
        }
        return Long.parseLong(digits.substring(p));
    }
}

class Solution {
    int findMaxConsecutiveBits(int target, int[] bits) {
        int left = 0;
        int nonTarget = 0;
        int ans = 0;

        for (int right = 0; right < bits.length; right++) {
            if (bits[right] != target) {
                nonTarget++;
            }
            while (nonTarget > 1) {
                if (bits[left] != target) {
                    nonTarget--;
                }
                left++;
            }
            ans = Math.max(ans, right - left + 1);
        }
        return ans;
    }
}

import java.util.*;
class Solution {
    int[] quickSort(int[] numbers) {
        int n = numbers.length;
        int[] preMax = new int[n];
        int[] sufMin = new int[n];

        preMax[0] = numbers[0];
        for (int i = 1; i < n; i++) {
            preMax[i] = Math.max(preMax[i - 1], numbers[i]);
        }
        sufMin[n - 1] = numbers[n - 1];
        for (int i = n - 2; i >= 0; i--) {
            sufMin[i] = Math.min(sufMin[i + 1], numbers[i]);
        }

        List<Integer> list = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            boolean leftOk = (i == 0) || preMax[i - 1] <= numbers[i];
            boolean rightOk = (i == n - 1) || sufMin[i + 1] >= numbers[i];
            if (leftOk && rightOk) {
                list.add(numbers[i]);
            }
        }

        int[] res = new int[list.size()];
        for (int i = 0; i < list.size(); i++) {
            res[i] = list.get(i);
        }
        return res;
    }
}

class Solution {
    long getLongestNum(long num) {
        if (num == 0) {
            return 1;
        }

        long best = 0;
        long cur = 0;
        long prev = -1;

        while (num > 0) {
            long bit = num & 1L;
            if (bit == prev) {
                cur++;
            } else {
                cur = 1;
                prev = bit;
            }
            best = Math.max(best, cur);
            num >>= 1;
        }
        return best;
    }
}

    int getMaxSendNumber(int cap, int[] bills, int[] priorities) {
        Map<Integer, List<Integer>> map = new TreeMap<>();
        for (int i = 0; i < bills.length; i++) {
            map.computeIfAbsent(priorities[i], k -> new ArrayList<>()).add(bills[i]);
        }

        int count = 0;
        for (List<Integer> list : map.values()) {
            Collections.sort(list);
            int sum = 0;
            for (int x : list) {
                sum += x;
            }

            if (sum <= cap) {
                cap -= sum;
                count += list.size();
            } else {
                for (int x : list) {
                    if (x <= cap) {
                        cap -= x;
                        count++;
                    } else {
                        break;
                    }
                }
                break;
            }
        }
        return count;
    }
}

        int n = inventory.length;
        Integer[] idx = new Integer[n];
        for (int i = 0; i < n; i++) {
            idx[i] = i;
        }

        Arrays.sort(idx, (a, b) -> {

        });

        double remain = demand;
        double total = 0.0;

        for (int k = 0; k < n && remain > 0; k++) {
            int i = idx[k];
            double take = Math.min(remain, inventory[i]);
            total += take * (price[i] / inventory[i]);
            remain -= take;
        }

        return String.format(java.util.Locale.ROOT, "%.2f", total);
    }
}

【软件认证】四则运算求值
import java.util.*;

class Solution {
    String calculate(String expression) {
        Deque<Long> stack = new ArrayDeque<>();
        long num = 0;
        char op = '+';
        int i = 0;
        int n = expression.length();

        boolean firstIsNegative = n > 0 && expression.charAt(0) == '-';
        if (firstIsNegative) {
            op = '-';
            i = 1;
        }

        while (i < n) {
            char c = expression.charAt(i);
            if (Character.isDigit(c)) {
                num = num * 10 + (c - '0');
            } else {
                if (!apply(stack, op, num)) {
                    return "error";
                }
                op = c;
                num = 0;
            }
            i++;
        }
        if (!apply(stack, op, num)) {
            return "error";
        }

        long sum = 0;
        while (!stack.isEmpty()) {
            sum += stack.pop();
        }
        return String.valueOf(sum);
    }

    private boolean apply(Deque<Long> stack, char op, long num) {
        switch (op) {
            case '+':
                stack.push(num);
                break;
            case '-':
                stack.push(-num);
                break;
            case '*':
                stack.push(stack.pop() * num);
                break;
            case '/':
                if (num == 0) {
                    return false;
                }
                stack.push(stack.pop() / num);
                break;
            default:
                break;
        }
        return true;
    }
}

class Solution {
    List<MyRecord> getPhoneRecords(List<Operation> operations) {
        Set<String> exact = new HashSet<>();
        List<String> prefixes = new ArrayList<>();
        Map<String, MyRecord> map = new LinkedHashMap<>();  

        for (Operation op : operations) {
            if (op.opType == 'W') {
                if (op.number.endsWith("*")) {
                    prefixes.add(op.number.substring(0, op.number.length() - 1));
                } else {
                    exact.add(op.number);
                }
            } else {  // 'C'
                boolean allow = exact.contains(op.number);
                if (!allow) {
                    for (String p : prefixes) {
                        if (op.number.startsWith(p)) {
                            allow = true;
                            break;
                        }
                    }
                }
                MyRecord r = map.get(op.number);
                if (r == null) {
                    r = new MyRecord();
                    r.number = op.number;
                    map.put(op.number, r);
                }
                if (allow) {
                    r.accepted++;
                } else {
                    r.refused++;
                }
            }
        }

        return new ArrayList<>(map.values());
    }
}

                return Integer.compare(weights[a], weights[b]);

            return Integer.compare(a, b);
        });

        for (int i = 0; i < n; i++) {
            res[i] = idx[i] + 1;
        }
        return res;
    }

import java.util.*;

class Solution {
    int getMinSum(int[] array1, int[] array2, int pairTarget) {
        int n = array1.length;
        int m = array2.length;

        PriorityQueue<int[]> heap = new PriorityQueue<>(
                (x, y) -> Integer.compare(array1[x[0]] + array2[x[1]],
                                          array1[y[0]] + array2[y[1]]));

        for (int i = 0; i < n; i++) {
            heap.offer(new int[]{i, 0});
        }

        int sum = 0;
        for (int k = 0; k < pairTarget && !heap.isEmpty(); k++) {
            int[] cur = heap.poll();
            int i = cur[0];
            int j = cur[1];
            sum += array1[i] + array2[j];
            if (j + 1 < m) {
                heap.offer(new int[]{i, j + 1});
            }
        }
        return sum;
    }
}

    String recombine(List<DataFragment> dataFragments, int dstType) {
        Map<Integer, String> contentMap = new HashMap<>();   
        Map<Integer, Integer> endMap = new HashMap<>();     

        for (DataFragment f : dataFragments) {
            if (f.dataType != dstType) {
                continue;
            }
            contentMap.putIfAbsent(f.index, f.data);
            endMap.put(f.index, f.endFlag);
        }

        if (contentMap.isEmpty()) {
            return "NA";
        }

        int maxSeq = Collections.max(contentMap.keySet());

        if (contentMap.size() != maxSeq + 1) {
            return "NA";
        }
        if (endMap.get(maxSeq) != 1) {
            return "NA";
        }

        StringBuilder sb = new StringBuilder();
        for (int i = 0; i <= maxSeq; i++) {
            sb.append(contentMap.get(i));
        }
        return sb.toString();
    }
}

class DataFragment {
    int dataType;
    int index;
    int endFlag;
    int len;
    String data;
    public String toString() {
        return String.format(java.util.Locale.ROOT, "[%d, %d, %d, %d, \"%s\"]", dataType, index, endFlag, len, data);
    }
}

class Solution {
    String transpose(String inputStr) {
        int len = inputStr.length();
        int n = (int) Math.round(Math.sqrt(len));
        if (n * n != len) {
            return "ERROR";
        }

        StringBuilder sb = new StringBuilder(len);
        for (int c = 0; c < n; c++) {
            for (int r = 0; r < n; r++) {
                sb.append(inputStr.charAt(r * n + c));
            }
        }
        return sb.toString();
    }
}

class Solution {
    List<String> regularID(List<String> cardIds) {
        Set<String> set = new TreeSet<>();

        for (String raw : cardIds) {
            String s = raw.replace(" ", "");
            if (s.isEmpty() || s.length() > 9) {
                continue;
            }

            char first = s.charAt(0);
            if (!Character.isLetter(first)) {
                continue;
            }

            boolean ok = true;
            boolean hasDigit = false;
            for (int i = 1; i < s.length(); i++) {
                if (!Character.isDigit(s.charAt(i))) {
                    ok = false;
                    break;
                }
                hasDigit = true;
            }
            if (!ok || !hasDigit) {
                continue;
            }

            String digits = s.substring(1);
            StringBuilder sb = new StringBuilder();
            sb.append(Character.toLowerCase(first));
            for (int i = digits.length(); i < 8; i++) {
                sb.append('0');
            }
            sb.append(digits);
            set.add(sb.toString());
        }

        return new ArrayList<>(set);
    }
}

class Solution {
    int getTheMinWithColMaxs(int[][] matrix) {
        int n = matrix.length;
        int m = matrix[0].length;
        int result = Integer.MAX_VALUE;

        for (int j = 0; j < m; j++) {
            int colMax = matrix[0][j];
            for (int i = 1; i < n; i++) {
                if (matrix[i][j] > colMax) {
                    colMax = matrix[i][j];
                }
            }
            result = Math.min(result, colMax);
        }

class Solution {

        boolean[] ok = new boolean[10];
        for (char c : numStr.toCharArray()) {
            ok[c - '0'] = true;
        }

        int cnt = 0;
        for (int v = 0; v <= 255; v++) {
            String s = String.valueOf(v);
            boolean valid = true;
            for (int i = 0; i < s.length(); i++) {
                if (!ok[s.charAt(i) - '0']) {
                    valid = false;
                    break;
                }
            }
            if (valid) {
                cnt++;
            }
        }

        long c = cnt;
        return c * c * c * c;
    }

class Solution {
    int getLongestVowelStrLen(String input) {
        int best = 0;
        int cur = 0;
        for (int i = 0; i < input.length(); i++) {
            if (isVowel(input.charAt(i))) {
                cur++;
                best = Math.max(best, cur);
            } else {
                cur = 0;
            }
        }
        return best;
    }

    private boolean isVowel(char c) {
        return "aeiouAEIOU".indexOf(c) >= 0;
    }
}
